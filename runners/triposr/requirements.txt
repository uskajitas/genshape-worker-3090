# (see SETUP.md → "Per-runner venv setup" → "Step 3: NVTX3 patch").
#
# Setup (full guide in SETUP.md):
#   py -3.11 -m venv .venv
#   .\.venv\Scripts\Activate.ps1
#   # ⚠ run from a Developer Command Prompt with v14.39 toolset:
#   #   "C:\Program Files (x86)\Microsoft Visual Studio\2022\BuildTools\
#   #    VC\Auxiliary\Build\vcvarsall.bat" x64 -vcvars_ver=14.39
#   pip install --index-url https://download.pytorch.org/whl/cu121 \
#       torch==2.5.1 torchvision==0.20.1
#   # Patch torch's cuda.cmake — see SETUP.md
#   pip install -r requirements.txt
#   git clone https://github.com/VAST-AI-Research/TripoSR.git \
#       C:\projects\ai\triposr\TripoSR
#   # Weights (gated; needs HF_TOKEN + license accepted on HF):
#   huggingface-cli download stabilityai/TripoSR \
#       config.yaml model.ckpt --local-dir C:\projects\ai\triposr
